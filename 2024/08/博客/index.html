先部署本地环境

  • 先下载hugo地址：hugo下载地址

  • 下载windows本地x64.exe文件。如果不同系统对应下载

  • 下载完成解压出hugo.exe文件

  • 在文件目录中输入cmd—->输入hugo new site test命令意思是创建一个hugo框架。

  • 目录结构说明

  • . ├── archetypes # 存放文章模板 ├── config.toml # 简单的配置文件 ├── content # 存放文章 ├── data # 存放生成静态页面时的配置文件 ├── layouts # 存放页面布局的模板 ├── static # 存放图片等静态内容 └── themes # 存放下载的主题

  • 输入hugo server -D在浏览器输入localhost:1313

添加博客运行时间

在layouts/partials/footer/custom.html里添加以下JS代码，其中s1是网站创建日期

<!-- 博客运行时间 -->
<script>
    let s1 = '2024-8-10'; //website start date
    s1 = new Date(s1.replace(/-/g, "/"));
    let s2 = new Date();
    let timeDifference = s2.getTime() - s1.getTime();

    let days = Math.floor(timeDifference / (1000 * 60 * 60 * 24));
    let hours = Math.floor((timeDifference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
    let minutes = Math.floor((timeDifference % (1000 * 60 * 60)) / (1000 * 60));

    let result = days + "天" + hours + "小时" + minutes + "分钟" ;
    document.getElementById('runningdays').innerHTML = result;
</script>

在layouts/partials/footer/footer.html里添加一下js代码注意添加位置

<!-- 博客运行时间 -->

</section>

统计发布多少文章和字数

在layouts/partials/footer/footer.html里增加以下代码

<!-- 博客统计 -->
<section class="totalcount">
    {{$scratch := newScratch}}
    {{ range (where .Site.Pages "Kind" "page") }}
        {{$scratch.Add "total" .WordCount}}
    {{ end }}
    
    {{$var := $scratch.Get "total"}}  <!-- 获取 total 值 -->
    {{$var = div $var 100.0}}         <!-- total 除以 100 -->
    {{$var = math.Ceil $var}}         <!-- 向上取整 -->
    {{$var = div $var 10.0}}          <!-- 再除以 10，获取带小数的数字 -->
    
    发表了 {{ len (where .Site.RegularPages "Section" "post") }} 篇文章 · 
    共 {{$var}}k字
</section>

// 樱花类
function Sakura(x, y, s, r, fn) {
    this.x = x; // 樱花的 x 坐标
    this.y = y; // 樱花的 y 坐标
    this.s = s; // 樱花的大小缩放因子
    this.r = r; // 樱花的旋转角度
    this.fn = fn; // 包含更新 x、y 和旋转角度的函数对象
}

// 绘制樱花
Sakura.prototype.draw = function(cxt) {
    cxt.save(); // 保存当前绘图状态
    var xc = 40 * this.s / 4; // 计算缩放后的樱花尺寸
    cxt.translate(this.x, this.y); // 将绘图上下文移动到樱花的位置
    cxt.rotate(this.r); // 旋转绘图上下文
    cxt.drawImage(img, 0, 0, 40 * this.s, 40 * this.s); // 绘制樱花图像
    cxt.restore(); // 恢复绘图状态
}

// 更新樱花的位置和状态
Sakura.prototype.update = function() {
    this.x = this.fn.x(this.x, this.y); // 更新 x 坐标
    this.y = this.fn.y(this.y, this.y); // 更新 y 坐标
    this.r = this.fn.r(this.r); // 更新旋转角度
    // 如果樱花超出屏幕边界，则重新生成其位置、大小和旋转角度
    if(this.x > window.innerWidth || this.x < 0 || this.y > window.innerHeight || this.y < 0) {
        this.r = getRandom('fnr'); // 重新生成旋转角度
        if(Math.random() > 0.4) {
            this.x = getRandom('x'); // 重新生成 x 坐标
            this.y = 0; // 重新生成 y 坐标
            this.s = getRandom('s'); // 重新生成大小
            this.r = getRandom('r'); // 重新生成旋转角度
        } else {
            this.x = window.innerWidth; // 重新生成 x 坐标
            this.y = getRandom('y'); // 重新生成 y 坐标
            this.s = getRandom('s'); // 重新生成大小
            this.r = getRandom('r'); // 重新生成旋转角度
        }
    }
}

// 樱花列表类
SakuraList = function() {
    this.list = []; // 初始化一个空的樱花列表
}

// 将樱花添加到列表中
SakuraList.prototype.push = function(sakura) {
    this.list.push(sakura);
}

// 更新列表中所有樱花的状态
SakuraList.prototype.update = function() {
    for(var i = 0, len = this.list.length; i < len; i++) {
        this.list[i].update();
    }
}

// 绘制列表中所有樱花
SakuraList.prototype.draw = function(cxt) {
    for(var i = 0, len = this.list.length; i < len; i++) {
        this.list[i].draw(cxt);
    }
}

// 获取指定索引的樱花
SakuraList.prototype.get = function(i) {
    return this.list[i];
}

// 生成随机值或函数
function getRandom(option) {
    var ret, random;
    switch(option) {
        case 'x':
            ret = Math.random() * window.innerWidth; // 生成随机的 x 坐标
            break;
        case 'y':
            ret = Math.random() * window.innerHeight; // 生成随机的 y 坐标
            break;
        case 's':
            ret = Math.random(); // 生成随机的大小缩放因子
            break;
        case 'r':
            ret = Math.random() * 6; // 生成随机的旋转角度
            break;
        case 'fnx':
            random = -0.5 + Math.random() * 1; // 生成随机值用于 x 坐标更新函数
            ret = function(x, y) {
                return x + 0.5 * random - 1.7; // 返回更新后的 x 坐标
            };
            break;
        case 'fny':
            random = 1.5 + Math.random() * 0.7; // 生成随机值用于 y 坐标更新函数
            ret = function(x, y) {
                return y + random; // 返回更新后的 y 坐标
            };
            break;
        case 'fnr':
            random = Math.random() * 0.03; // 生成随机值用于旋转角度更新函数
            ret = function(r) {
                return r + random; // 返回更新后的旋转角度
            };
            break;
    }
    return ret;
}

// 启动樱花动画
function startSakura() {
    // 兼容不同浏览器的 requestAnimationFrame 方法
    requestAnimationFrame = window.requestAnimationFrame ||
        window.mozRequestAnimationFrame ||
        window.webkitRequestAnimationFrame ||
        window.msRequestAnimationFrame ||
        window.oRequestAnimationFrame;

    var canvas = document.createElement('canvas'), // 创建画布
        cxt;
    staticx = true;
    canvas.height = window.innerHeight; // 设置画布高度
    canvas.width = window.innerWidth; // 设置画布宽度
    canvas.setAttribute('style', 'position: fixed;left: 0;top: 0;pointer-events: none;'); // 设置画布样式
    canvas.setAttribute('id', 'canvas_sakura'); // 设置画布 ID
    document.getElementsByTagName('body')[0].appendChild(canvas); // 将画布添加到文档中
    cxt = canvas.getContext('2d'); // 获取绘图上下文
    var sakuraList = new SakuraList(); // 创建樱花列表
    // 生成 50 个樱花
    for(var i = 0; i < 50; i++) {
        var sakura, randomX, randomY, randomS, randomR, randomFnx, randomFny;
        randomX = getRandom('x'); // 获取随机的 x 坐标
        randomY = getRandom('y'); // 获取随机的 y 坐标
        randomR = getRandom('r'); // 获取随机的旋转角度
        randomS = getRandom('s'); // 获取随机的大小缩放因子
        randomFnx = getRandom('fnx'); // 获取随机的 x 坐标更新函数
        randomFny = getRandom('fny'); // 获取随机的 y 坐标更新函数
        randomFnR = getRandom('fnr'); // 获取随机的旋转角度更新函数
        sakura = new Sakura(randomX, randomY, randomS, randomR, {
            x: randomFnx,
            y: randomFny,
            r: randomFnR
        });
        sakura.draw(cxt); // 绘制樱花
        sakuraList.push(sakura); // 将樱花添加到列表中
    }
    // 创建动画循环
    stop = requestAnimationFrame(function() {
        cxt.clearRect(0, 0, canvas.width, canvas.height); // 清除画布
        sakuraList.update(); // 更新樱花状态
        sakuraList.draw(cxt); // 绘制樱花
        stop = requestAnimationFrame(arguments.callee); // 递归调用创建动画效果
    });
}

// 当窗口大小改变时，调整画布大小
window.onresize = function() {
    var canvasSakura = document.getElementById('canvas_sakura');
    canvasSakura.width = window.innerWidth; // 更新画布宽度
    canvasSakura.height = window.innerHeight; // 更新画布高度
}

// 图像加载完成后，启动樱花动画
img.onload = function() {
    startSakura();
}

// 启动或停止樱花动画
function stopp() {
    if(staticx) {
        var child = document.getElementById("canvas_sakura");
        child.parentNode.removeChild(child); // 移除画布
        window.cancelAnimationFrame(stop); // 取消动画帧请求
        staticx = false; // 更新状态
    } else {
        startSakura(); // 重新启动樱花动画
    }
}
